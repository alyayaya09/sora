<script data-purpose="page-interactions">
    document.addEventListener('DOMContentLoaded', () => {
      // Like buttons interactivity
      const likeButtons = document.querySelectorAll('button[aria-label="Suka"], button[aria-label="Favorit"]');
      likeButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          btn.classList.toggle('text-rose-400');
        });
      });

      // Carousel Hover & Sticky Active State Interactivity
      const itemWrappers = document.querySelectorAll('#batik-carousel-container .carousel-card-item');
      let currentActiveWrapper = itemWrappers[2]; // Default active: Card 3 (Wajan Peleburan Motif-Motif)

      function setActiveCard(targetWrapper) {
        if (!targetWrapper || currentActiveWrapper === targetWrapper) return;

        // Nonaktifkan kartu aktif sebelumnya
        if (currentActiveWrapper) {
          const prevCard = currentActiveWrapper.querySelector('.interactive-card');
          currentActiveWrapper.classList.remove('is-active-wrapper');
          currentActiveWrapper.classList.add('is-inactive-wrapper');
          if (prevCard) {
            prevCard.classList.remove('is-active', 'ring-2', 'ring-amber-400/50');
            prevCard.classList.add('is-inactive');
          }
        }

        // Aktifkan kartu baru yang sedang di-hover
        currentActiveWrapper = targetWrapper;
        const newCard = targetWrapper.querySelector('.interactive-card');
        targetWrapper.classList.remove('is-inactive-wrapper');
        targetWrapper.classList.add('is-active-wrapper');
        if (newCard) {
          newCard.classList.remove('is-inactive');
          newCard.classList.add('is-active', 'ring-2', 'ring-amber-400/50');
        }
      }

      itemWrappers.forEach((wrapper) => {
        wrapper.addEventListener('mouseenter', () => {
          setActiveCard(wrapper);
        });
      });

      // Catatan: Event mouseleave pada container sengaja tidak di-reset agar kartu terakhir yang disentuh tetap aktif (sticky active state)

      // IntersectionObserver for entrance & exit scroll animations from left & right
      const animElements = document.querySelectorAll('.scroll-animate, .carousel-card-anim-wrapper');
      
      const scrollObserverOptions = {
        root: null,
        rootMargin: '0px 0px -60px 0px',
        threshold: [0, 0.15]
      };

      const scrollObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            // Meluncur masuk ke dalam posisi normal
            entry.target.classList.add('is-visible');
          } else {
            // Meluncur keluar kembali ke arah luar (exit animation)
            entry.target.classList.remove('is-visible');
          }
        });
      }, scrollObserverOptions);

      animElements.forEach(el => scrollObserver.observe(el));
    });
  </script>
